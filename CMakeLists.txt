cmake_minimum_required(VERSION 3.20)
project(cuda_nbody LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# CUDA is optional; it must be enabled before targets are created so they pick up
# CMAKE_CUDA_ARCHITECTURES. Without it the GPU entry points are stubs.
option(NB_USE_CUDA "Build the CUDA kernels if a CUDA compiler is found" ON)
set(NB_HAVE_CUDA OFF)
if(NB_USE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES 70 75 80 86)
    endif()
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    set(NB_HAVE_CUDA ON)
  endif()
endif()

add_library(nbody src/cpu.cpp)
target_include_directories(nbody PUBLIC include)
find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(nbody PUBLIC OpenMP::OpenMP_CXX)
endif()
if(NB_HAVE_CUDA)
  target_sources(nbody PRIVATE src/gpu.cu)
  target_compile_options(nbody PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-use_fast_math>)
  message(STATUS "CUDA enabled: ${CMAKE_CUDA_COMPILER}, architectures ${CMAKE_CUDA_ARCHITECTURES}")
else()
  target_sources(nbody PRIVATE src/gpu_stub.cpp)
  message(STATUS "CUDA disabled or not found: building the CPU path only")
endif()

add_executable(nbody_bench app/bench.cpp)
target_link_libraries(nbody_bench PRIVATE nbody)
add_executable(collide app/collide.cpp)
target_link_libraries(collide PRIVATE nbody)

enable_testing()
add_executable(test_cpu tests/test_cpu.cpp)
target_link_libraries(test_cpu PRIVATE nbody)
add_test(NAME cpu COMMAND test_cpu)
add_executable(test_gpu tests/test_gpu.cpp)
target_link_libraries(test_gpu PRIVATE nbody)
add_test(NAME gpu COMMAND test_gpu)
set_tests_properties(gpu PROPERTIES SKIP_RETURN_CODE 77)
